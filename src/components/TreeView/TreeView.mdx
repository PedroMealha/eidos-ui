# TreeView

Hierarchical list with expand/collapse nodes. Supports single and multi-selection, custom label rendering, guide lines, and controlled expansion state.

## Usage

```tsx
import { TreeView } from '@pmealha/eidos-ui';
import type { TreeNode } from '@pmealha/eidos-ui';

const data: TreeNode[] = [
  {
    id: 'src',
    label: 'src',
    children: [
      { id: 'index', label: 'index.ts' },
    ],
  },
  { id: 'readme', label: 'README.md' },
];

<TreeView data={data} defaultExpandedIds={['src']} />
```
