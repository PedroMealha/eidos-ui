# Tabs

Content switcher with a horizontal row of labelled tab triggers. Supports underline (default), enclosed, and pills visual styles.

## Usage

```tsx
import { Tabs, Tab, TabPanel } from '@pmealha/eidos-ui';

<Tabs defaultValue="overview">
  <Tab value="overview">Overview</Tab>
  <Tab value="analytics">Analytics</Tab>
  <TabPanel value="overview">Overview content</TabPanel>
  <TabPanel value="analytics">Analytics content</TabPanel>
</Tabs>
```
