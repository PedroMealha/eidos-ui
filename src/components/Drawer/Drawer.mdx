# Drawer

Slide-in panel anchored to a screen edge. Use it for settings panels, detail views, or any content that should overlay the page without a full navigation change.

## Usage

```tsx
import { Drawer } from 'eidos-ui';
import { useState } from 'react';

function Example() {
  const [isOpen, setIsOpen] = useState(false);
  return (
    <>
      <button onClick={() => setIsOpen(true)}>Open</button>
      <Drawer isOpen={isOpen} onClose={() => setIsOpen(false)} title="Settings">
        <p>Drawer content goes here.</p>
      </Drawer>
    </>
  );
}
```
