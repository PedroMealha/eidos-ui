@use '../../styles/mixins.scss' as *;
 
.eidos-pagelayout {
  box-sizing: border-box;
  height: 100%;
  width: 100%;
  display: grid;
  grid-template-areas:
    'navigation toolbar'
    'navigation content';
  grid-template-columns: auto 1fr;
  grid-template-rows: auto 1fr;
  background-color: var(--white);
 
  &__toolbar {
    grid-area: toolbar;
    border-bottom: 1px solid var(--gray-200);
  }
 
  &__navigation {
    grid-area: navigation;
    border-right: 1px solid var(--gray-200);
 
    // `navigation` is optional, and the region renders as an empty div when
    // it is omitted - at which point this border was the column's entire
    // width, leaving a stray 1px rule down the edge of a layout that is
    // documented as collapsing "to nothing". `:empty` matches because the
    // region has no child nodes at all in that case.
    &:empty {
      border-right: none;
    }
  }
 
  &__content {
    grid-area: content;
    display: flex;
    flex-direction: column;
    background-color: var(--gray-50);
    // A grid item's automatic minimum size is its content's size, so without
    // these the outer `1fr` content row can only grow: tall page content
    // stretched the whole layout past the viewport and scrolled the toolbar
    // and navigation rail away with it, and a wide table did the same
    // horizontally. With the row pinned, `overflow: auto` makes *this*
    // element the layout's only scrollport - the header, body and footer all
    // size to their own content and scroll together inside it.
    //
    // Deliberately not a grid with fixed `auto 1fr auto` rows: rows are
    // resolved against the scrollport height, so the body's track could never
    // grow past it and tall content overflowed its own area and was painted
    // over by the footer. Auto-placement made it worse - with `header`
    // omitted, every region shifted up a track and the footer inherited the
    // `1fr` leftover.
    min-height: 0;
    min-width: 0;
    overflow: auto;
  }
 
  &__header,
  &__footer {
    // Size to their own content, and never get squashed by a tall body.
    flex: 0 0 auto;
  }
 
  &__body {
    // Grows to push the footer down to the bottom edge on a short page, and
    // grows *past* the scrollport (rather than shrinking) on a long one.
    // `flex-shrink: 0`, plus the absence of any `min-height` floor, is what
    // turns tall content into scrollable overflow on `&__content` instead of
    // an overlap with the footer.
    flex: 1 0 auto;
  }
 
  &__footer {
    // Only the default `copyright` variant gets the extra vertical spacing - a
    // custom `component` is expected to fully own its own layout/spacing.
    // Vertical only: the left/right inset is the shared page gutter below, and
    // setting it here as well is what used to pin the footer's horizontal
    // padding, since this selector outranks the gutter's media queries.
    &.eidos-footer--copyright {
      padding-block: var(--spacing-md);
    }
  }
 
  // Vertical rhythm, per region. Horizontal insets are deliberately absent -
  // see the page gutter below.
  &__header,
  &__body,
  &__footer {
    padding-block: var(--spacing-xl) 0;
 
    @include media-down('xl') {
      padding-block: var(--spacing-lg) 0;
    }
 
    @include media-down('lg') {
      padding-block: var(--spacing-md) 0;
    }
 
    @include media-down('md') {
      padding-block: var(--spacing-sm) 0;
    }
  }
 
  // Overrides the shared `0` bottom padding above: the end of the page
  // content would otherwise sit flush against the footer with no breathing
  // room between the two.
  &__body {
    padding-block-end: var(--spacing-xl);
 
    @include media-down('xl') {
      padding-block-end: var(--spacing-lg);
    }
 
    @include media-down('lg') {
      padding-block-end: var(--spacing-md);
    }
 
    @include media-down('md') {
      padding-block-end: var(--spacing-sm);
    }
  }
}
 
// ============================================================================
// PAGE GUTTER - one left/right inset shared by every region
// ============================================================================
//
// The toolbar, header, body and footer all sit on the same vertical edges at
// every width. Three separate things used to stop that happening:
//
// 1. `Toolbar` brings its own `padding` and no breakpoints, so it stayed at
//    the widest inset while the body stepped down - 42px against 7px at a
//    700px viewport, measured.
// 2. The footer's `copyright` rule above is a two-class selector, so it
//    outranked the media queries in the ladder and never stepped down either.
// 3. **`--spacing-*` are `em`.** Even with both of those fixed the regions
//    would still not line up: the footer renders at `--font-size-sm`, so the
//    same `--spacing-xl` resolved to 36.75px there against the body's 42px.
//
// Hence `rem` here rather than the spacing tokens - a page gutter has to be
// one distance, not a multiple of whatever text each region happens to
// contain. The values are the spacing scale at the library's own 14px root
// (`global.scss`), so the desktop gutter is exactly what it was.
//
// Written as a descendant selector on purpose: at `.eidos-pagelayout__toolbar`
// alone this would tie with `.eidos-toolbar`'s own padding and lose on source
// order, because `Toolbar.scss` is imported after this file. The extra class
// makes the layout win regardless of import order, which is the right
// outcome - a layout owns the page's edges, and a standalone `Toolbar` keeps
// its own padding untouched.
.eidos-pagelayout {
  .eidos-pagelayout__toolbar,
  .eidos-pagelayout__header,
  .eidos-pagelayout__body,
  .eidos-pagelayout__footer {
    padding-inline: 3rem;
 
    @include media-down('xl') {
      padding-inline: 1.5rem;
    }
 
    @include media-down('lg') {
      padding-inline: 1rem;
    }
 
    @include media-down('md') {
      padding-inline: 0.5rem;
    }
  }
}
